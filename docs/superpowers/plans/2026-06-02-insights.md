## Task 2: Migration — `v_monthly_pl`

**Files:** Create `supabase/migrations/0011_insights.sql`

- [ ] **Step 1: Write the migration**

```sql
-- =============================================================
-- Insights phase 1: monthly P&L building blocks for trend charts.
-- Read-only view; no change to posting, RPCs, or existing reports.
-- =============================================================
create or replace view v_monthly_pl with (security_invoker = on) as
select
  le.org_id,
  date_trunc('month', le.date)::date as month,
  coalesce(sum(case when a.system_key = 'sales' then le.credit - le.debit else 0 end), 0)::bigint as sales,
  coalesce(sum(case when a.system_key = 'cogs'  then le.debit - le.credit else 0 end), 0)::bigint as cogs,
  coalesce(
    sum(case when a.system_key = 'sales' then le.credit - le.debit else 0 end)
  - sum(case when a.system_key = 'cogs'  then le.debit - le.credit else 0 end), 0)::bigint as gross_profit,
  coalesce(sum(case when a.group_id = 4 then le.credit - le.debit else 0 end), 0)::bigint as income,
  coalesce(sum(case when a.group_id = 5 then le.debit - le.credit else 0 end), 0)::bigint as expense,
  coalesce(
    sum(case when a.group_id = 4 then le.credit - le.debit else 0 end)
  - sum(case when a.group_id = 5 then le.debit - le.credit else 0 end), 0)::bigint as net_profit
from ledger_entries le
join accounts a on a.id = le.account_id
group by le.org_id, date_trunc('month', le.date);

grant select on v_monthly_pl to authenticated;
```

```ts
export type MonthlyPL = {
  month: string            // 'YYYY-MM-DD' (first of month)
  sales: number; cogs: number; gross_profit: number
  income: number; expense: number; net_profit: number
}
export function useMonthlyPL(orgId: string | null, months = 12) {
  return useQuery({
    queryKey: ['monthly_pl', orgId, months],
    enabled: !!orgId,
    queryFn: async (): Promise<MonthlyPL[]> => {
      const { data, error } = await supabase
        .from('v_monthly_pl')
        .select('month, sales, cogs, gross_profit, income, expense, net_profit')
        .eq('org_id', orgId)
        .order('month', { ascending: true })
      if (error) throw error
      const rows = (data ?? []) as MonthlyPL[]
      return rows.slice(-months)
    },
  })
}
```

## Task 5: Pure insight builders

**Files:** Create `src/features/insights/insights.ts`

- [ ] **Step 1: Write the pure functions**

```ts
import type { MonthlyPL } from '@/hooks/queries'

export type InsightTone = 'pos' | 'neg' | 'neutral'
export type Insight = { id: string; label: string; value: string; detail?: string; tone: InsightTone; arrow?: 'up' | 'down' }

/** Month-over-month % change; null when there is no comparable prior value. */
export function momPct(current: number, previous: number): number | null {
  if (!previous) return null
  return Math.round(((current - previous) / Math.abs(previous)) * 1000) / 10
}

type BuildInput = {
  monthly: MonthlyPL[]
  cashBank: number          // paise
  receivablesTotal: number  // paise
  receivablesCount: number
  fmt: (paise: number) => string   // formatINR
}

/**
 * Compose plain-language insight cards from already-fetched data.
 * Cards self-omit when their data isn't meaningful.
 */
export function buildInsights({ monthly, cashBank, receivablesTotal, receivablesCount, fmt }: BuildInput): Insight[] {
  const out: Insight[] = []
  const last = monthly[monthly.length - 1]
  const prev = monthly[monthly.length - 2]

  if (last) {
    const pct = prev ? momPct(last.sales, prev.sales) : null
    out.push({
      id: 'sales',
      label: 'Sales this month',
      value: fmt(last.sales),
      detail: pct === null ? undefined : `${pct >= 0 ? '↑' : '↓'} ${Math.abs(pct)}% vs last month`,
      tone: pct === null ? 'neutral' : pct >= 0 ? 'pos' : 'neg',
      arrow: pct === null ? undefined : pct >= 0 ? 'up' : 'down',
    })
    out.push({
      id: 'profit',
      label: 'Profit this month',
      value: fmt(last.net_profit),
      tone: last.net_profit >= 0 ? 'pos' : 'neg',
    })
  }

  if (receivablesTotal > 0) {
    out.push({
      id: 'receivables',
      label: 'Customers owe you',
      value: fmt(receivablesTotal),
      detail: `across ${receivablesCount} ${receivablesCount === 1 ? 'party' : 'parties'}`,
      tone: 'neutral',
    })
  }

  out.push({ id: 'cash', label: 'Cash + bank', value: fmt(cashBank), tone: 'neutral' })
  return out
}
```

```tsx
import {
  ResponsiveContainer, ComposedChart, Bar, Line, XAxis, YAxis, Tooltip, CartesianGrid, Legend,
} from 'recharts'
import type { MonthlyPL } from '@/hooks/queries'
import { formatINR } from '@/lib/money'

const MONTHS = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec']
const label = (iso: string) => {
  const d = new Date(iso)
  return `${MONTHS[d.getMonth()]} ${String(d.getFullYear()).slice(2)}`
}
// charts plot rupees (paise/100) for readable axes
const toRupees = (paise: number) => Math.round(paise / 100)
const tip = (v: number) => formatINR(v * 100)

const axis = { stroke: '#94a3b8', fontSize: 12, tickLine: false, axisLine: false } as const
const grid = { stroke: '#e8ebf0', vertical: false } as const
const tooltipStyle = {
  border: '1px solid #e8ebf0', borderRadius: 12, fontSize: 13,
  boxShadow: '0 4px 12px rgba(15,23,42,.06)',
}

export function SalesProfitChart({ data }: { data: MonthlyPL[] }) {
  const rows = data.map((m) => ({ name: label(m.month), Sales: toRupees(m.sales), Profit: toRupees(m.net_profit) }))
  return (
    <ResponsiveContainer width="100%" height={260}>
      <ComposedChart data={rows} margin={{ top: 8, right: 8, left: 8, bottom: 0 }}>
        <CartesianGrid {...grid} />
        <XAxis dataKey="name" {...axis} />
        <YAxis {...axis} width={48} tickFormatter={(v) => (v >= 1000 ? `${v / 1000}k` : `${v}`)} />
        <Tooltip formatter={(v: number) => tip(v)} contentStyle={tooltipStyle} cursor={{ fill: 'rgba(79,70,229,.06)' }} />
        <Legend iconType="circle" wrapperStyle={{ fontSize: 12 }} />
        <Bar dataKey="Sales" fill="#4f46e5" radius={[6, 6, 0, 0]} maxBarSize={34} />
        <Line dataKey="Profit" stroke="#059669" strokeWidth={2.5} dot={false} />
      </ComposedChart>
    </ResponsiveContainer>
  )
}

export function IncomeExpenseChart({ data }: { data: MonthlyPL[] }) {
  const rows = data.map((m) => ({ name: label(m.month), Income: toRupees(m.income), Expense: toRupees(m.expense) }))
  return (
    <ResponsiveContainer width="100%" height={240}>
      <ComposedChart data={rows} margin={{ top: 8, right: 8, left: 8, bottom: 0 }}>
        <CartesianGrid {...grid} />
        <XAxis dataKey="name" {...axis} />
        <YAxis {...axis} width={48} tickFormatter={(v) => (v >= 1000 ? `${v / 1000}k` : `${v}`)} />
        <Tooltip formatter={(v: number) => tip(v)} contentStyle={tooltipStyle} cursor={{ fill: 'rgba(79,70,229,.06)' }} />
        <Legend iconType="circle" wrapperStyle={{ fontSize: 12 }} />
        <Bar dataKey="Income" fill="#4f46e5" radius={[6, 6, 0, 0]} maxBarSize={22} />
        <Bar dataKey="Expense" fill="#cbd5e1" radius={[6, 6, 0, 0]} maxBarSize={22} />
      </ComposedChart>
    </ResponsiveContainer>
  )
}
```

```tsx
import { ArrowUpRight, ArrowDownRight } from 'lucide-react'
import { useAuth } from '@/hooks/useAuth'
import { useMonthlyPL, useDashboard, useAged } from '@/hooks/queries'
import { formatINR } from '@/lib/money'
import { Card } from '@/components/ui/Card'
import { PageHeader } from '@/components/ui/PageHeader'
import { EmptyState } from '@/components/ui/EmptyState'
import { LineChart } from 'lucide-react'
import { buildInsights, type Insight } from './insights'
import { SalesProfitChart, IncomeExpenseChart } from './charts'

const toneText: Record<Insight['tone'], string> = {
  pos: 'text-pos', neg: 'text-neg', neutral: 'text-ink',
}

function InsightCard({ ins }: { ins: Insight }) {
  return (
    <div className="rounded-2xl border border-line bg-surface p-4 card-elev">
      <p className="text-sm font-medium text-muted">{ins.label}</p>
      <p className={`num mt-1 text-2xl font-bold tracking-tight ${toneText[ins.tone]}`}>{ins.value}</p>
      {ins.detail && (
        <p className={`mt-1 flex items-center gap-1 text-xs ${ins.tone === 'pos' ? 'text-pos' : ins.tone === 'neg' ? 'text-neg' : 'text-muted'}`}>
          {ins.arrow === 'up' && <ArrowUpRight size={13} />}
          {ins.arrow === 'down' && <ArrowDownRight size={13} />}
          {ins.detail}
        </p>
      )}
    </div>
  )
}

export function InsightsPage() {
  const { currentOrgId } = useAuth()
  const { data: monthly = [], isLoading } = useMonthlyPL(currentOrgId)
  const { data: dash } = useDashboard(currentOrgId)
  const { data: aged = [] } = useAged(currentOrgId, 'receivable')

  const receivablesTotal = aged.reduce((s, r) => s + r.outstanding, 0)
  const cashBank = (dash?.cash_balance ?? 0) + (dash?.bank_balance ?? 0)
  const insights = buildInsights({
    monthly, cashBank, receivablesTotal, receivablesCount: aged.length, fmt: formatINR,
  })

  const hasData = monthly.some((m) => m.sales !== 0 || m.expense !== 0)

  return (
    <div className="space-y-6">
      <PageHeader title="Insights" description="How your business is doing — trends and plain-language highlights, at a glance." />

      {isLoading ? (
        <div className="grid grid-cols-2 gap-3 lg:grid-cols-4">
          {Array.from({ length: 4 }).map((_, i) => <div key={i} className="h-24 animate-pulse rounded-2xl border border-line bg-surface" />)}
        </div>
      ) : !hasData ? (
        <Card>
          <EmptyState icon={LineChart} title="Not enough activity yet"
            description="Record a few sales, purchases, and expenses and your trends and highlights will appear here." />
        </Card>
      ) : (
        <>
          <div className="grid grid-cols-2 gap-3 rise lg:grid-cols-4">
            {insights.map((ins) => <InsightCard key={ins.id} ins={ins} />)}
          </div>

          <Card>
            <h2 className="mb-3 text-base font-semibold text-ink">Sales &amp; profit · last 12 months</h2>
            <SalesProfitChart data={monthly} />
          </Card>

          <Card>
            <h2 className="mb-3 text-base font-semibold text-ink">Income vs expense</h2>
            <IncomeExpenseChart data={monthly} />
          </Card>
        </>
      )}
    </div>
  )
}
```
